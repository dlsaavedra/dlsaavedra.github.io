# Secuencia de tiempos para predicción
times = np.linspace(0, datos['Meses'].max(), 100)

# Función de supervivencia teórica para cada modelo
surv_exp = exp_model.survival_function_at_times(times)
surv_lnorm = lnorm_model.survival_function_at_times(times)
surv_weibull = weibull_model.survival_function_at_times(times)

# Crear DataFrame para ggplot (equivalente en Python)
df_curves = pd.DataFrame({
    'time': np.tile(times, 3),
    'surv': np.concatenate([surv_exp, surv_lnorm, surv_weibull]),
    'model': np.repeat(['Exponencial', 'Log-Normal', 'Weibull'], len(times))
})

# Gráfico
plt.figure(figsize=(10, 6))
sns.lineplot(data=df_curves, x='time', y='surv', hue='model', linewidth=2)

# Agregar la curva de Kaplan-Meier
# Extraer la curva de supervivencia y los intervalos de confianza
km_df = km.survival_function_.reset_index()
km_df.columns = ['time', 'surv']

# Extraer intervalos de confianza y unirlos
ci_df = km.confidence_interval_.reset_index()
ci_df.columns = ['time', 'lower', 'upper']

# Combinar todo en un único dataframe
km_df = pd.merge(km_df, ci_df, on='time')

plt.step(km_df['time'], km_df['surv'], color='black', linewidth=2, where='post')
plt.fill_between(km_df['time'], km_df['lower'], km_df['upper'], color='gray', alpha=0.3)

plt.title("Comparación de Modelos Paramétricos vs Kaplan-Meier")
plt.xlabel("Tiempo (Meses)")
plt.ylabel("Probabilidad de Supervivencia S(t)")
plt.legend(title="Modelo Paramétrico", loc="lower left")
plt.show()
