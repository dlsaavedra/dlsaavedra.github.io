# 6. Pruebas de bondad de ajuste
# Extraer predictores lineales
fits_normal = np.log(aft_normal.predict_median(data_expanded).values)
fits_weibull =  np.log(aft_weibull.predict_median(data_expanded).values)

# Residuos: log(Ti) - η_i / σ
logT = np.log(data_expanded['Hours'].values)
scale_normal = np.exp(aft_normal.params_["sigma_"][0])
scale_weibull = np.exp(-aft_weibull.params_["rho_"][0])

resids_normal = (logT - fits_normal) / scale_normal
resids_weibull = (logT - fits_weibull) / scale_weibull

# Función de ayuda para graficar residuos
def plot_residuals(resids, evento, theoretical_sf, label, ylim):
    kmf = KaplanMeierFitter()
    kmf.fit(resids, event_observed=evento)

    # Graficar estimación KM
    plt.figure(figsize=(8, 5))
    kmf.plot_survival_function(ci_show=False, label='KM de residuos')
    
    # Superponer función de supervivencia teórica
    xx = np.linspace(min(resids), max(resids), 100)
    plt.plot(xx, theoretical_sf(xx), 'r-', lw=2, label='Función teórica')
    
    plt.title(f'Análisis de residuos: {label}')
    plt.xlabel('Residuales')
    plt.ylabel('Probabilidad de supervivencia')
    plt.ylim(ylim)
    plt.legend()
    plt.grid(True)
    plt.tight_layout()
    plt.show()

# Residuales para modelo Normal (función de supervivencia: 1 - Φ(x))
plot_residuals(resids_normal, data_expanded['Evento'], lambda x: 1 - norm.cdf(x), 'Log-Normal', (0.5, 1))

# Residuales para modelo Weibull (función de supervivencia: exp(-exp(x)))
plot_residuals(resids_weibull, data_expanded['Evento'], lambda x: np.exp(-np.exp(x)), 'Weibull', (0.9, 1))

# 1. Configurar parámetros iniciales
temp_vals = [10]  # Valores de temperatura a evaluar
tseq = np.linspace(1, 35000, 1000)  # Secuencia de tiempos de 1 a 35,000 horas

# Asumimos que tienes los modelos AFT ya ajustados:
# aft_weibull y aft_gaussian son objetos de modelos ajustados

# 2. Predecir supervivencia Weibull
weibull_scale = np.exp(-aft_weibull.params_["rho_"][0])
weibull_coef = np.array([aft_weibull.params_['lambda_']['Intercept'], 
                        aft_weibull.params_['lambda_']['Temperature']])

weibull_surv = np.zeros((len(tseq), len(temp_vals)))

for i, temp in enumerate(temp_vals):
    # Calcular predictor lineal (1 para intercepto + valor de temperatura)
    lp = np.dot(weibull_coef, [1, temp])
    scale = np.exp(lp)
    # Calcular probabilidad de supervivencia Weibull
    weibull_surv[:, i] = 1 - weibull_min.cdf(tseq, 1/weibull_scale, scale=scale)

# 3. Predecir supervivencia Normal (Log-Normal)
normal_scale = np.exp(aft_normal.params_["sigma_"][0])
normal_coef = np.array([aft_normal.params_["mu_"]['Intercept'], 
                       aft_normal.params_["mu_"]['Temperature']])

normal_surv = np.zeros((len(tseq), len(temp_vals)))

for i, temp in enumerate(temp_vals):
    # Calcular predictor lineal
    lp = np.dot(normal_coef, [1, temp])
    # Calcular puntuaciones z
    z = (np.log(tseq) - lp) / normal_scale
    # Calcular probabilidad de supervivencia Normal
    normal_surv[:, i] = 1 - norm.cdf(z)

# 4. Crear DataFrames para visualización
df_weibull = pd.DataFrame({
    'Tiempo': np.tile(tseq, len(temp_vals)),
    'Supervivencia': weibull_surv.flatten(),
    'Modelo': 'Weibull',
    'Temperatura': np.repeat(temp_vals, len(tseq))
})

df_normal = pd.DataFrame({
    'Tiempo': np.tile(tseq, len(temp_vals)),
    'Supervivencia': normal_surv.flatten(),
    'Modelo': 'Normal',
    'Temperatura': np.repeat(temp_vals, len(tseq))
})

df_modelos = pd.concat([df_weibull, df_normal])

# 5. Visualización con matplotlib
plt.figure(figsize=(10, 6))
